%%%%%% QUERY 1
% EXPLANATION: Checks the cost of three nights (the 17th, 18th and 19th) at the
% Hilton in Saskatoon, where the room costs 180 + 180 + 195 = 555.
stayCost(the_hilton, saskatoon, 17, 20, Cost).


%%%%%% QUERY 2
% EXPLANATION: Checks the cost of two nights (the 5th and 6th) at the Holiday
% Inn in Toronto, where the room costs 140 + 140 = 280.
stayCost(holiday_inn, toronto, 5, 7, Cost).


%%%%%% QUERY 3
% EXPLANATION: With Hotel as a variable, finds the hotel in Toronto that has a
% room on both the 6th and the 7th. Only four_seasons qualifies, at 9 + 182 = 191.
stayCost(Hotel, toronto, 6, 8, Cost).


%%%%%% QUERY 4
% EXPLANATION: With both Hotel and City as variables, finds every hotel/city
% that has a room on both the 12th and the 13th: best_western in montreal
% (80 + 80 = 160) and marriot in london (120 + 120 = 240).
stayCost(Hotel, City, 12, 14, Cost).


%%%%%% QUERY 5
% EXPLANATION: Checks that a zero-night stay fails, because the check-in date
% must be strictly before the check-out date.
stayCost(holiday_inn, toronto, 6, 6, Cost).


%%%%%% QUERY 6
% EXPLANATION: Checks that the query fails when the hotel has no room on every
% required night: four_seasons has a room on the 25th but not on the 26th, so
% a stay from the 25th to the 27th is impossible.
stayCost(four_seasons, toronto, 25, 27, Cost).


%%%%%% QUERY 7
% EXPLANATION: Finds every 2-flight round trip that leaves Toronto on the 12th
% and returns on the 14th: Toronto->Montreal->Toronto (370) and
% Toronto->London->Toronto (510).
roundTrip(toronto, 12, 14, 2, Cost).


%%%%%% QUERY 8
% EXPLANATION: Checks that the 2-flight Toronto->Montreal->Toronto trip on the
% 12th/14th costs exactly 370 (100 flight + 160 hotel + 110 flight).
roundTrip(toronto, 12, 14, 2, 370).


%%%%%% QUERY 9
% EXPLANATION: With City as a variable, finds all 2-flight round trips that
% leave on the 12th and return on the 14th from any city. Only Toronto has
% such trips.
roundTrip(City, 12, 14, 2, Cost).


%%%%%% QUERY 10
% EXPLANATION: Checks that no legal 3-flight round trip from Toronto fits
% between the 12th and the 14th, so the query fails.
roundTrip(toronto, 12, 14, 3, Cost).


%%%%%% QUERY 11
% EXPLANATION: Checks that a query with NumFlights < 2 fails, since every trip
% must go somewhere other than the starting city and then return.
roundTrip(toronto, 12, 14, 1, Cost).


%%%%%% QUERY 12
% EXPLANATION: Checks that the query fails when no single hotel is available
% for the whole layover: Toronto->Montreal on the 12th and back on the 21st
% would need a Montreal hotel for the 12th through the 20th, which the KB
% does not have (Montreal only has rooms on the 12th and 13th).
roundTrip(toronto, 12, 21, 2, Cost).
